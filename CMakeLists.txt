cmake_minimum_required(VERSION 3.25)
project(UltraHDR VERSION 2.0.2 LANGUAGES CXX)
include(GNUInstallDirs)
find_package(JPEG REQUIRED)
find_package(Threads REQUIRED)

option(UHDR_BUILD_STATIC "Build the static library" ON)
option(UHDR_BUILD_SHARED "Build the shared library" ON)
file(GLOB_RECURSE image_io_sources CONFIGURE_DEPENDS third_party/image_io/src/*.cc)
file(GLOB core_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/lib/src/*.cpp")
add_library(image_io STATIC ${image_io_sources})
target_include_directories(image_io PUBLIC third_party/image_io/includes
    PRIVATE third_party/image_io/src/modp_b64 third_party/image_io/src/modp_b64/modp_b64)
set_target_properties(image_io PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(core OBJECT ${core_sources})
set(uhdr_arch "${CMAKE_SYSTEM_PROCESSOR}")
if(CMAKE_CXX_COMPILER_ARCHITECTURE_ID)
    set(uhdr_arch "${CMAKE_CXX_COMPILER_ARCHITECTURE_ID}")
endif()
if(APPLE AND CMAKE_OSX_ARCHITECTURES)
    set(uhdr_arch "${CMAKE_OSX_ARCHITECTURES}")
endif()
string(TOLOWER "${uhdr_arch}" uhdr_arch)
if(uhdr_arch MATCHES "^(aarch64|arm64|arm|armv[7-9].*)$")
    target_sources(core PRIVATE lib/src/dsp/arm/editorhelper_neon.cpp lib/src/dsp/arm/gainmapmath_neon.cpp)
    target_compile_definitions(core PRIVATE UHDR_ENABLE_INTRINSICS)
endif()
target_compile_features(core PUBLIC cxx_std_17)
target_compile_definitions(core PRIVATE UHDR_WRITE_XMP UHDR_WRITE_ISO)
target_include_directories(core PUBLIC . lib/include)
target_link_libraries(core PRIVATE image_io JPEG::JPEG Threads::Threads)
set_target_properties(core PROPERTIES POSITION_INDEPENDENT_CODE ON)

function(add_uhdr target type)
    add_library(${target} ${type} $<TARGET_OBJECTS:core>)
    target_link_libraries(${target} PRIVATE image_io JPEG::JPEG Threads::Threads)
    target_include_directories(${target} PUBLIC
        "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>"
        "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")
endfunction()
if(UHDR_BUILD_STATIC)
    add_uhdr(uhdr-static STATIC)
endif()
if(UHDR_BUILD_SHARED)
    add_uhdr(uhdr SHARED)
    set_target_properties(uhdr PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 2
        WINDOWS_EXPORT_ALL_SYMBOLS ON)
endif()

if(WIN32 AND TARGET uhdr)
    enable_language(RC)
    configure_file(cmake/LibraryVersion.rc.in LibraryVersion.rc @ONLY)
    target_sources(uhdr PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/LibraryVersion.rc")
endif()
